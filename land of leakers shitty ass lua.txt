local function cleanTripESP()
    for _, h in pairs(tripHighlights) do pcall(function() h:Destroy() end) end

local function isTripmine(obj)
    if not obj then return false end
    local name = obj.Name:lower()
    return name:find("tripmine") or name:find("subspace") or name:find("mine")
end
local function addTripESP(obj)
    if not obj:IsA("Model") and not obj:IsA("BasePart") then return end
    if not isTripmine(obj) then return end
    local highlight = Instance.new("SelectionBox")
    highlight.Color3 = tripColorESP
    highlight.LineThickness = 0.05
    highlight.SurfaceTransparency = 0.6
    highlight.SurfaceColor3 = tripColorESP
    highlight.Adornee = obj
    highlight.Parent = workspace
    table.insert(tripHighlights, highlight)
    local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
    if part then
        local billboard = Instance.new("BillboardGui")
        billboard.AlwaysOnTop = true
        billboard.Size = UDim2.new(0, 120, 0, 40)
        billboard.StudsOffset = Vector3.new(0, 3, 0)
        billboard.Adornee = part
        billboard.Parent = part
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.TextColor3 = tripColorESP
        label.TextStrokeTransparency = 0
        label.TextScaled = true
        label.Font = Enum.Font.GothamBold
        label.Text = "TRIPMINE"
        label.Parent = billboard
        local distConn
        distConn = RunService.RenderStepped:Connect(function()
            if not part or not part.Parent then distConn:Disconnect() pcall(function() billboard:Destroy() end) return end
            local char = lp.Character
            if char then
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hrp then label.Text = "TRIPMINE\n" .. math.floor((hrp.Position - part.Position).Magnitude) .. " studs" end
            end
        end)
        table.insert(tripBillboards, billboard)
    end
    obj.AncestryChanged:Connect(function()
        if not obj:IsDescendantOf(workspace) then pcall(function() highlight:Destroy() end) end
    end)
end
local function scanTripmines()
    for _, obj in pairs(workspace:GetDescendants()) do if isTripmine(obj) then addTripESP(obj) end end
end

local VOID_X_MAX = 450000
local VOID_Y_MAX = 590000
local VOID_Z_MAX = 230000
local VoidIntensity = 1.0
local function betterRandom(mi, ma, dmi, dma)
    local val = math.random(mi, ma)
    local attempts = 0
    repeat
        val = math.random(mi, ma)
        attempts = attempts + 1
    until val < dmi or val > dma or attempts > 20
    return val
end
local function getVoidPos()
    local xi = math.floor(VOID_X_MAX * VoidIntensity)
    local yi = math.floor(VOID_Y_MAX * VoidIntensity)
    local zi = math.floor(VOID_Z_MAX * VoidIntensity)
    local xe = math.floor(xi * 0.5)
    local ye = math.floor(yi * 0.5)
    local ze = math.floor(zi * 0.5)
    return Vector3.new(betterRandom(-xi, xi, -xe, xe), betterRandom(0, yi, 0, ye), betterRandom(-zi, zi, -ze, ze))
end
local function getVoidVel()
    local xi = math.floor(VOID_X_MAX * VoidIntensity)
    local yi = math.floor(VOID_Y_MAX * VoidIntensity)
    local zi = math.floor(VOID_Z_MAX * VoidIntensity)
    return Vector3.new(math.random(-xi, xi), math.random(-yi, yi), math.random(-zi, zi))
end
local function lockCamera() lockedCameraC = workspace.CurrentCamera.CFrame end
local function applyCameraLock() if lockedCameraC then workspace.CurrentCamera.CFrame = lockedCameraC end end
local function startVoidSpam()
    if voidHBConn then voidHBConn:Disconnect() voidHBConn = nil end
    lockCamera()
    voidHBConn = RunService.Heartbeat:Connect(function()
        pcall(function()
            local char = lp.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end
            voidHRPRef = hrp
            savedClientC = hrp.CFrame
            savedClientV = hrp.AssemblyLinearVelocity
            savedClientVA = hrp.AssemblyAngularVelocity
            local vPos = getVoidPos()
            hrp.CFrame = CFrame.new(vPos) * CFrame.Angles(math.pi, math.pi, math.pi)
            hrp.AssemblyLinearVelocity = getVoidVel()
            hrp.AssemblyAngularVelocity = getVoidVel()
        end)
    end)
    RunService:BindToRenderStep("VoidSync", Enum.RenderPriority.First.Value, function()
        if voidHRPRef and savedClientC then pcall(function() voidHRPRef.CFrame = savedClientC voidHRPRef.AssemblyLinearVelocity = savedClientV voidHRPRef.AssemblyAngularVelocity = savedClientVA end) end
        applyCameraLock()
    end)
end
local function stopVoidSpam()
    if voidHBConn then voidHBConn:Disconnect() voidHBConn = nil end
    pcall(function() RunService:UnbindFromRenderStep("VoidSync") end)
    voidHRPRef = nil
    lockedCameraC = nil
end

end

local function setFastKatanaDeflect()
    task.wait(0.01)
    for _, v in pairs(getgc(true)) do
        if type(v) == "table" then
            if rawget(v, "DeflectCooldown") then rawset(v, "DeflectCooldown", 0.01) end
            if rawget(v, "DeflectDuration") then rawset(v, "DeflectDuration", 0.01) end
        end
    end
    task.wait(0.01)
    pcall(function()
        local Modules = ReplicatedStorage:FindFirstChild("Modules")
        if not Modules then return end
        local lib = Modules:FindFirstChild("ItemLibrary") or Modules:FindFirstChild("ItemLibrary", true)
        if not lib then return end
        local ItemLibrary = require(lib)
        if ItemLibrary and ItemLibrary.Items then
            for _, item in pairs(ItemLibrary.Items) do
                if item.Name and string.find(item.Name:lower(), "katana") then
                    item.DeflectCooldown = 0.01
                    item.DeflectDuration = 0.01
                end
            end
        end
    end)
    task.wait(0.01)
    pcall(function()
        local MeleeModule = safeRequire(lp.PlayerScripts.Modules.ItemTypes.Melee)
        if not MeleeModule then return end
        local oldNew = MeleeModule.new
        MeleeModule.new = function(...)
            local self = oldNew(...)
            if self and self.Info and self.Info.Name and string.find(self.Info.Name:lower(), "katana") then
                self.Info.DeflectCooldown = 0.01
                self.Info.DeflectDuration = 0.8
            end
            return self
        end
    end)
end

local function isPlayerInGame(player)
    if not tpConfig.onlyInGame then return true end
    if player == lp then return false end
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    if not char:IsDescendantOf(workspace) then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local myChar = lp.Character
    if myChar then
        local myHRP = myChar:FindFirstChild("HumanoidRootPart")
        if myHRP then
            local dist = (myHRP.Position - hrp.Position).Magnitude
            if dist > 5000 then return false end
        end
    end
    return true
end

local function teleportToPlayer(player)
    if not isPlayerInGame(player) then Library:Notify(player.Name .. " is in lobby — skipped!", 3) return false end
    local char = player.Character
    local myChar = lp.Character
    if not char or not myChar then return false end
    local tHRP = char:FindFirstChild("HumanoidRootPart")
    local myHRP = myChar:FindFirstChild("HumanoidRootPart")
    if not tHRP or not myHRP then return false end
    myHRP.CFrame = CFrame.new(tHRP.Position + Vector3.new(0, 5, 0))
    if tpConfig.stayOnTop then startStayOnTop(player) end
    Library:Notify("Teleported to " .. player.Name, 2)
    return true
end
local hitboxConfig = { enabled = false, size = 10, part = "HitboxHead", transparency = 0.8, showESP = true }
local originalSizes = {}
local hitboxConn = nil
local function getHitboxPart(player)
    local char = workspace:FindFirstChild(player.Name)
    if not char then return nil end
    local hitbox = char:FindFirstChild(hitboxConfig.part)
    if not hitbox then return nil end
    if not hitbox:IsA("BasePart") then return nil end
    return hitbox
end
local function storeOriginalSize(player, hitbox)
    if originalSizes[player.Name] then return end
    local attrSize = hitbox:GetAttribute("OriginalSize")
    if attrSize then
        originalSizes[player.Name] = attrSize
        return
    end
    local sizeVal = hitbox:FindFirstChild("OriginalSize")
    if sizeVal then
        originalSizes[player.Name] = sizeVal.Value
        return
    end
    originalSizes[player.Name] = hitbox.Size
end
local function expandHitbox(player)
    if player == lp then return end
    local hitbox = getHitboxPart(player)
    if not hitbox then return end
    storeOriginalSize(player, hitbox)
    local newSize = Vector3.new(hitboxConfig.size, hitboxConfig.size, hitboxConfig.size)
    pcall(function()
        hitbox.Size = newSize
        if hitboxConfig.showESP then
            hitbox.Transparency = hitboxConfig.transparency
            hitbox.Color = Color3.fromRGB(255, 50, 50)
            hitbox.Material = Enum.Material.Neon
            hitbox.CanCollide = false
            hitbox.CastShadow = false
        else
            hitbox.Transparency = 1
        end
    end)
end
local function restoreHitbox(player)
    local hitbox = getHitboxPart(player)
    if not hitbox then return end
    local original = originalSizes[player.Name]
    if not original then return end
    pcall(function()
        hitbox.Size = original
        hitbox.Transparency = 1
        hitbox.Material = Enum.Material.SmoothPlastic
        hitbox.Color = Color3.fromRGB(255, 255, 255)
    end)
    originalSizes[player.Name] = nil
end
local function expandAll()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= lp then pcall(function() expandHitbox(player) end) end
    end
end
local function restoreAll()
    for _, player in pairs(Players:GetPlayers()) do pcall(function() restoreHitbox(player) end) end
    originalSizes = {}
end
Players.PlayerAdded:Connect(function(player)
    player.CharacterAdded:Connect(function()
        if hitboxConfig.enabled and player ~= lp then
            task.wait(0.5)
            pcall(function() expandHitbox(player) end)

    end)
end)

    if player ~= lp then
        player.CharacterAdded:Connect(function()
            if hitboxConfig.enabled then
                task.wait(0.5)
                pcall(function() originalSizes[player.Name] = nil expandHitbox(player) end)
            end
        end)
    end
end
